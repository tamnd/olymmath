$$
=\frac{n+1}{2}-\frac{\sin (n+1) \beta \cos (2 \alpha+n \beta)}{2 \sin \beta} ;
$$
$$
\begin{aligned}
\text { (2) } & \sin \alpha \sin (\alpha+\beta)+\sin (\alpha+\beta) \sin (\alpha+2 \beta)+\cdots \\
& +\sin [\alpha+(n-1) \beta] \sin (\alpha+n \beta) \\
= & \frac{n \cos \beta}{2}-\frac{\sin n \beta \cos (2 \alpha+n \beta)}{2 \sin \beta}
\end{aligned}
$$

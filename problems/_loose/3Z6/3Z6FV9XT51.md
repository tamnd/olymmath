$$
\left\{\begin{array}{l}
a_{n+1}=3 a_{n}+b_{n}-4 \\
b_{n+1}=2 a_{n}+2 b_{n}+2
\end{array} (n \geqslant 0), a_{0}=4, b_{0}=0 .\right.
$$

$$
\begin{array}{lllllllll}
\text { value } & x_{i} & -2 & 1 & 2 & 3 & 4 & 5 \\
\text { frequency } & n_{i} & 2 & 2 & 2 & 2 & 2 & 1
\end{array}
$$

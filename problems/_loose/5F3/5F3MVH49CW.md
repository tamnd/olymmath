$$
\begin{aligned}
& 20^{3} x_{1}+21^{3} x_{2}+\cdots+49^{3} x_{30}=13 \\
& 21^{3} x_{1}+22^{3} x_{2}+\cdots+50^{3} x_{30}=1 \\
& 22^{3} x_{1}+23^{3} x_{2}+\cdots+51^{3} x_{30}=19
\end{aligned}
$$

$$
\frac{a^2(a^3 + b^3)}{a^2 + ab + b^2} + \frac{b^2(b^3 + c^3)}{b^2 + bc + c^2} + \frac{c^2(c^3 + a^3)}{c^2 + ca + a^2} \ge 2abc \text{ holds.}
$$

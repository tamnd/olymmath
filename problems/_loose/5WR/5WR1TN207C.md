$$
\sqrt{1+\sqrt{a}}+\sqrt{1+\sqrt{a+\sqrt{a^{2}}}}+\ldots+\sqrt{1+\sqrt{a+\sqrt{a^{2}+\ldots+\sqrt{a^{n}}}}}<n a
$$

$$
\begin{aligned}
(x^2 + 1)y &= z^2 + 1 \\
(y^2 + 1)z &= x^2 + 1 \\
(z^2 + 1)x &= y^2 + 1.
\end{aligned}
(\text{Ukraine 2017})
$$

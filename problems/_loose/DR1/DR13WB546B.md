\begin{tabular}{cc}
$A= \begin{pmatrix} 1 & 2 & 3 & \cdots & n\\ n & 1 & 2 & \cdots & n - 1\\ n - 1 & n & 1 & \cdots & n - 2\\ \cdots & \cdots & \cdots & \cdots & \cdots\\2 & 3 & 4 & \cdots & 1 \end{pmatrix}$

\end{tabular}

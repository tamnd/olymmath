$
\textbf{(A)}\ \dfrac{5}{\sqrt 3}
\textbf{(B)}\ 2
\textbf{(C)}\ \dfrac{\sqrt 5}2
\textbf{(D)}\ 3
\textbf{(E)}\ \dfrac{\sqrt 7}2

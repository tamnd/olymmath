$$
x_{1}^{2 n}+\left(x_{1} x_{2}\right)^{2 n}+\ldots+\left(x_{1} x_{2} \ldots x_{n}\right)^{2 n}=1
$$

$$
x_{1}^{2 n+1}+\left(x_{1} x_{2}\right)^{2 n+1}+\ldots+\left(x_{1} x_{2} \ldots x_{n}\right)^{2 n+1}=1
$$

$
\textbf{(A)}\ \sqrt{17}
\textbf{(B)}\ \sqrt[3]{30}
\textbf{(C)}\ \sqrt{26}-1
\textbf{(D)}\ \sqrt {30}
\textbf{(E)}\ \sqrt [3]{10}
$

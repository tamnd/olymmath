$$
\begin{aligned}
& \left|z_{1}\right|+\left|z_{2}\right|+\left|z_{3}\right|+\left|z_{1}+z_{2}+z_{3}\right| \\
\geqslant & \left|z_{1}+z_{2}\right|+\left|z_{2}+z_{3}\right|+\left|z_{3}+z_{1}\right|
\end{aligned}
$$

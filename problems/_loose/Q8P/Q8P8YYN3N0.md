$$
\begin{gathered}
x^{4}+y^{4}+z^{4}=13 \\
x^{3} y^{3} z+y^{3} z^{3} x+z^{3} x^{3} y=6 \sqrt{3} \\
x^{3} y z+y^{3} z x+z^{3} x y=5 \sqrt{3} ?
\end{gathered}
$$

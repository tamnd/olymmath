$$
\begin{gathered}
x+y+z=\frac{a b}{c}+\frac{b c}{a}+\frac{c a}{b} \\
c x+a y+b z=a b+b c+c a \\
c^{2} x+a^{2} y+b^{2} z=3 a b c
\end{gathered}
$$

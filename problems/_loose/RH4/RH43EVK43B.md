| ---: | ---: | ---: | ---: | ---: | ---: | ---: |
| II. | 30007, | 207, | $60407 ;$ | 7, | 60207, | 30407, |
| III. | 4, | 500, | $405 ;$ | 5, | 400, | 504, |
| IV. | 6, | 350, | $164 ;$ | 14, | 150, | 356. |

[ (1) (a_1x + b_1)^2 + (a_2x + b_2)^2, \]

[ (2) (b_1x + c_1)^2 + (b_2x + c_2)^2 \]

[ (3) (c_1x + a_1)^2 + (c_2x + a_2)^2 \]

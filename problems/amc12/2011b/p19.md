\textbf{(A)}\ \frac{51}{101} \textbf{(B)}\ \frac{50}{99} \textbf{(C)}\ \frac{51}{100} \textbf{(D)}\ \frac{52}{101} \textbf{(E)}\ \frac{13}{25}

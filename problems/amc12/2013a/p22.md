\textbf{(A)} \ \frac{8}{25} \textbf{(B)} \ \frac{33}{100} \textbf{(C)} \ \frac{7}{20} \textbf{(D)} \ \frac{9}{25} \textbf{(E)} \ \frac{11}{30}

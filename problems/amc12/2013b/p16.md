\textbf{(A)}\ 0 \textbf{(B)}\ \frac{1}{2} \textbf{(C)}\ \frac{\sqrt{5}-1}{2} \textbf{(D)}\ \frac{\sqrt{5}+1}{2} \textbf{(E)}\ \sqrt{5}

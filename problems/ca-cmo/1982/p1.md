In the diagram, $OB_i$ is parallel and equal in length to $A_i A_{i + 1}$ for $i = 1$ , 2, 3, and 4 ( $A_5 = A_1$ ). Show that the area of $B_1 B_2 B_3 B_4$ is twice that of $A_1 A_2 A_3 A_4$ .

[asy]
unitsize(1 cm);

pair O;
pair[] A, B;

O = (0,0);
A[1] = (0.5,-3);
A[2] = (2,0);
A[3] = (-0.2,0.5);
A[4] = (-1,0);
B[1] = A[2] - A[1];
B[2] = A[3] - A[2];
B[3] = A[4] - A[3];
B[4] = A[1] - A[4];

draw(A[1]--A[2]--A[3]--A[4]--cycle);
draw(B[1]--B[2]--B[3]--B[4]--cycle);
draw(O--B[1]);
draw(O--B[2]);
draw(O--B[3]);
draw(O--B[4]);

label(" $A_1$ ", A[1], S);
label(" $A_2$ ", A[2], E);
label(" $A_3$ ", A[3], N);
label(" $A_4$ ", A[4], W);
label(" $B_1$ ", B[1], NE);
label(" $B_2$ ", B[2], W);
label(" $B_3$ ", B[3], SW);
label(" $B_4$ ", B[4], S);
label(" $O$ ", O, E);
[/asy]
